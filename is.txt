// Add debug logging for rejected tools
fn accept_tool_call(tc: ToolCall, tool_names: &[&str]) -> Option<ToolCall> {
    if tool_names.is_empty() {
        return Some(tc); // Allow any tool if none specified
    }
    if tool_names.contains(&tc.name.as_str()) {
        return Some(tc);
    }
    // Fuzzy match: strip "_run" suffix, case-insensitive
    let normalized = tc.name.to_lowercase().replace("_run", "");
    if tool_names.iter().any(|n| n.to_lowercase().replace("_run", "") == normalized) {
        debug!(original = %tc.name, matched = ?tool_names, "Tool name fuzzy-matched");
        return Some(tc);
    }
    debug!(tool = %tc.name, available = ?tool_names, "Tool call rejected: name mismatch");
    None
}
Also: move spinner stripping to happen after JSON extraction, not before, to avoid corrupting quoted strings.

fn accept_tool_call(tc: ToolCall, tool_names: &[&str]) -> Option<ToolCall> {
    if tool_names.is_empty() {
        debug!(tool = %tc.name, "Tool accepted: no restrictions");
        return Some(tc);
    }
    // Exact match
    if tool_names.contains(&tc.name.as_str()) {

        return Some(tc);

Add helper detect_tool_early in src/qwen.rs (append before #[cfg(test)]):

rust

/// Incremental tool detection: returns Some if a complete tool-call JSON is found in prefix.
/// Uses brace-matching to avoid false positives on partial JSON.
pub fn detect_tool_early(text: &str, tool_defs: &[Value]) -> Option<ToolCall> {
    let tool_names: Vec<&str> = tool_defs.iter().filter_map(|t| t["name"].as_str()).collect();
    // Try to find a complete JSON object containing "tool" key
    let mut start = 0;
    while let Some(rel) = text[start..].find("{\"tool\"") {
        let json_start = start + rel;
        if let Some(end) = find_json_object_end(text, json_start) {
            let candidate = &text[json_start..=end];
            if let Some(tc) = try_parse_tool_json(candidate) {
                return accept_tool_call(tc, &tool_names);
            }
            start = end + 1;
        } else {
            break; // Incomplete JSON, wait for more text
        }
    }
    None
}
